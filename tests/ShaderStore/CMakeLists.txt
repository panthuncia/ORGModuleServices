cmake_minimum_required(VERSION 3.21)
project(ORGShaderStoreTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
include(../../cmake/ShaderStore.cmake)
add_executable(ORGShaderStoreTests ShaderStoreTests.cpp)
target_compile_features(ORGShaderStoreTests PRIVATE cxx_std_20)
target_link_libraries(ORGShaderStoreTests PRIVATE Threads::Threads ORGModuleServices::ShaderStore)
target_compile_definitions(ORGShaderStoreTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ORGShaderStoreTests PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>)
add_test(NAME ORGShaderStoreTests COMMAND ORGShaderStoreTests)
set_tests_properties(ORGShaderStoreTests PROPERTIES TIMEOUT 120)
